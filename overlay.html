// AIOverlay.js const AI_SERVICE_URL = "http://ТВОЙ_IP_СЕРВЕРА:8080/process"; export function initAIOverlay(currentConfig, botId) { const html = `

DeepSeek AI Admin

`; document.body.insertAdjacentHTML('beforeend', html); const box = document.getElementById('ai-box'); const btn = document.getElementById('ai-toggle-btn'); const sendBtn = document.getElementById('ai-send-btn'); const input = document.getElementById('ai-txt'); const status = document.getElementById('ai-status'); btn.onclick = () => { box.style.display = box.style.display === 'none' ? 'block' : 'none'; btn.style.transform = box.style.display === 'none' ? 'scale(1)' : 'scale(0.9)'; }; sendBtn.onclick = async () => { const prompt = input.value.strip(); if (!prompt) return; status.innerText = "⏳ Нейронка меняет реальность..."; sendBtn.disabled = true; sendBtn.style.opacity = "0.5"; try { const res = await fetch(AI_SERVICE_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: prompt, current_config: currentConfig, bot_id: botId }) }); if (!res.ok) throw new Error("Ошибка сервиса"); const data = await res.json(); status.innerText = "✅ Конфиг обновлен!"; console.log("Новый JSON:", data); // Важно: тут должна быть твоя функция обновления интерфейса админки if (window.onConfigUpdate) window.onConfigUpdate(data); } catch (e) { status.innerText = "❌ Ошибка: " + e.message; } finally { sendBtn.disabled = false; sendBtn.style.opacity = "1"; } }; }
